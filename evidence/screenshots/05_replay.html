INSERT INTO weather.weather_replay SELECT * FROM weather.weather_city
        WHERE city_id = 'moscow' AND toDate(observed_at) = '2024-10-05';

INSERT INTO weather.weather_replay SELECT * FROM weather.weather_city
        WHERE city_id = 'moscow' AND toDate(observed_at) = '2024-10-05' AND toHour(observed_at) < 6;

SELECT sum(observations) AS observations, sum(temp_count) AS temp_count,
        sum(temp_sum_centi) AS temp_sum_centi, sum(precip_sum_centi) AS precip_sum_centi
        FROM weather.daily_replay;
